  <script>
    const GAS_API_URL = "https://script.google.com/macros/s/AKfycby6wjxckZ0oaBkGCKxhZi6zCLuE8M8dv83GLqGOhf65R8wR7W4zHWv26Xdq4FdYz6k/exec";

    let adminPassword = "";
    let lastPendingCount = 0;
    let isFirstLoad = true;

    async function callGasApi(actionName, payload = {}) {
      try {
        const response = await fetch(GAS_API_URL, {
          method: "POST",
          body: JSON.stringify({ action: actionName, ...payload })
        });
        return await response.json();
      } catch (err) {
        return { success: false, message: "網路通訊失敗：" + err.toString() };
      }
    }

    async function login() {
      const pwd = document.getElementById('pwdInput').value.trim();
      const res = await callGasApi('verifyAdminPassword', { password: pwd });
      if (res && res.success) {
        adminPassword = pwd;
        document.getElementById('loginArea').style.display = 'none';
        document.getElementById('adminMain').style.display = 'block';
        loadAdminData();
        setInterval(loadAdminData, 4000);
      } else {
        alert('密碼錯誤！');
      }
    }

    function playBeepAndSpeech(msgText) {
      try {
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.type = 'sine';
        osc.frequency.setValueAtTime(880, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.3);

        if ('speechSynthesis' in window) {
          setTimeout(() => {
            const synth = window.speechSynthesis;
            const utter = new SpeechSynthesisUtterance(msgText);
            utter.lang = 'zh-TW';
            synth.speak(utter);
          }, 350);
        }
      } catch (e) {}
    }

    async function loadAdminData() {
      const data = await callGasApi('getKitchenAdminData', { password: adminPassword });
      if (!data || !data.success) return;

      if (!isFirstLoad && data.pendingCount > lastPendingCount) {
        const newestOrder = data.list.find(item => item.status === '出餐中');
        playBeepAndSpeech('收到新訂單，' + (newestOrder ? newestOrder.tableNo : ''));
      }

      lastPendingCount = data.pendingCount;
      isFirstLoad = false;

      document.getElementById('pendingCount').innerText = data.pendingCount + ' 單';
      document.getElementById('readyCount').innerText = data.readyCount + ' 單';
      document.getElementById('completedCount').innerText = data.completedCount + ' 單';
      document.getElementById('totalSales').innerText = 'NT$ ' + data.totalSales.toLocaleString();

      const tbody = document.getElementById('tableBody');
      if (data.list.length === 0) {
        tbody.innerHTML = '<tr><td colspan="6" style="text-align:center; padding: 20px;">目前無訂單資料</td></tr>';
        return;
      }

      tbody.innerHTML = '';
      data.list.forEach(item => {
        const tr = document.createElement('tr');
        let actionBtn = '';
        let badgeClass = 'badge-pending';

        if (item.status === '出餐中') {
          actionBtn = `<button class="btn-ready" onclick="updateStatus('${item.orderId}', '可取餐')">製作完成 (通知取餐)</button>`;
          badgeClass = 'badge-pending';
        } else if (item.status === '可取餐') {
          actionBtn = `<button class="btn-done" onclick="updateStatus('${item.orderId}', '取餐完成')">顧客已取餐</button>`;
          badgeClass = 'badge-ready';
        } else {
          actionBtn = `<span style="color:#aaa;">已結單</span>`;
          badgeClass = 'badge-done';
        }

        tr.innerHTML = `
          <td><span class="order-id-tag">${item.orderId}</span><br><span style="color:#888; font-size:11px;">${item.time}</span></td>
          <td><strong>${item.tableNo}</strong></td>
          <td>${item.items}${item.note ? '<br><span style="color:red;font-size:11px;">註:'+item.note+'</span>' : ''}</td>
          <td><strong style="color:#e74c3c;">NT$ ${item.amount}</strong></td>
          <td><span class="badge ${badgeClass}">${item.status}</span></td>
          <td>${actionBtn}</td>
        `;
        tbody.appendChild(tr);
      });
    }

    async function updateStatus(orderId, targetStatus) {
      await callGasApi('updateOrderStatusByAdmin', {
        orderId: orderId,
        password: adminPassword,
        targetStatus: targetStatus
      });
      loadAdminData();
    }
